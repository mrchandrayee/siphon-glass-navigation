siphon.js

// update the selected navigation item
const items = document
  .querySelectorAll('.nav-item');

items.forEach(item => {
  item.addEventListener('click', () => {
    items.forEach(link =>
      link.classList.remove('active'));
    item.classList.add('active');
  });
});
